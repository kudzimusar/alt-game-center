    function setGrade(grade) {
        currentGrade = grade;
        currentQuestions = [...quizData[grade]].sort(() => Math.random() - 0.5);
        qIndex = 0;
        isQuizActive = true;
        document.getElementById('grade-picker').classList.add('hidden');
        document.getElementById('quiz-ui').classList.remove('hidden');
        document.getElementById('grade-label').innerText = grade;
        loadQuestion();

    function loadQuestion() {
        const q = currentQuestions[qIndex];
        document.getElementById('t1').innerHTML = q.h1;
        document.getElementById('t2').innerHTML = q.h2;
        document.getElementById('t3').innerHTML = q.h3;
        document.querySelectorAll('.hint-card').forEach(c => c.classList.remove('visible'));
        document.getElementById('letter-hint-container').innerHTML = '';
        document.getElementById('answer-box').classList.add('hidden');
        document.getElementById('answer-img').removeAttribute('src');
        revealedLettersCount = 0;
        lucide.createIcons();
    }

    function revealHint(n) {
        document.getElementById(`hint-${n}`).classList.add('visible');
    }

    function revealLetter() {
        const answer = currentQuestions[qIndex].ans;
        if (revealedLettersCount < answer.length) {
            const char = answer[revealedLettersCount];
            const circle = document.createElement('div');
            circle.className = 'letter-circle';
            circle.innerText = char === ' ' ? '' : char.toUpperCase();
            if (char === ' ') circle.style.border = 'none';
            document.getElementById('letter-hint-container').appendChild(circle);
            revealedLettersCount++;
        }
    }

    async function showAnswer() {
        const q = currentQuestions[qIndex];
        document.getElementById('final-answer').innerText = q.ans;
        document.getElementById('answer-box').classList.remove('hidden');
        document.getElementById('loading-spinner').style.display = 'flex';
        
        try {
            const imgUrl = await fetchImagen(q.query, q.category);
            const imgEl = document.getElementById('answer-img');
            imgEl.src = imgUrl;
            imgEl.onload = () => document.getElementById('loading-spinner').style.display = 'none';
        } catch (error) {
            handleImageError();
        }
        revealHint(1); revealHint(2); revealHint(3);
    }

    async function fetchImagen(query, category) {
        const url = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`;
        const enhancedPrompt = `High quality, clear, bright photo of ${query}. Subject is a ${category}. Center focus, realistic, educational, no text.`;
        
        const response = await fetchWithRetry(url, {
            method: "POST",
            body: JSON.stringify({
                instances: { prompt: enhancedPrompt },
                parameters: { sampleCount: 1 }
            })
        });

        const data = await response.json();
        if (data.predictions && data.predictions[0].bytesBase64Encoded) {
            return `data:image/png;base64,${data.predictions[0].bytesBase64Encoded}`;
        }
        throw new Error("Gen failed");
    }

    async function fetchWithRetry(url, options, retries = 5) {
        let delay = 1000;
        for (let i = 0; i < retries; i++) {
            try {
                const response = await fetch(url, options);
                if (response.ok) return response;
            } catch (e) {}
            await new Promise(r => setTimeout(r, delay));
            delay *= 2;
        }
        throw new Error("Final fail");
    }

    function handleImageError() {
        if (!currentQuestions[qIndex]) return;
        const q = currentQuestions[qIndex];
        const fallback = `https://loremflickr.com/800/600/${encodeURIComponent(q.ans.toLowerCase())}?lock=${qIndex}`;
        document.getElementById('answer-img').src = fallback;
        document.getElementById('loading-spinner').style.display = 'none';
    }

    function nextQuestion() {
        qIndex = (qIndex + 1) % currentQuestions.length;
        loadQuestion();
    }

    function updateScore(team, val) {
        teamScores[team] += val;
        document.getElementById(`score${team}`).innerText = teamScores[team];
    }

    function goHome() {
        isQuizActive = false;
        document.getElementById('grade-picker').classList.remove('hidden');
        document.getElementById('quiz-ui').classList.add('hidden');
    }

    window.addEventListener('keydown', (e) => {
        const key = e.key.toLowerCase();
        if (isQuizActive) {
            if (key === '1') revealHint(1);
            if (key === '2') revealHint(2);
            if (key === '3') revealHint(3);
            if (key === 'l') revealLetter();
            if (key === ' ' || key === 'a') { e.preventDefault(); showAnswer(); }
            if (key === 'n') nextQuestion();
        }
        if (key === 'h') goHome();
    });

        document.getElementById('grade-picker').classList.add('hidden');
        document.getElementById('quiz-ui').classList.remove('hidden');
        
        const label = document.getElementById('grade-label');
        label.innerText = grade === 'JHS1' ? 'GRADE 1' : grade === 'JHS2' ? 'GRADE 2' : 'GRADE 3';
        label.className = `px-8 py-3 rounded-2xl text-white font-black text-xl shadow-inner gradient-${grade.toLowerCase()}`;
        
        loadQuestion();
    }

    function loadQuestion() {
        const q = currentQuestions[qIndex];
        document.getElementById('t1').innerHTML = q.h1;
        document.getElementById('t2').innerHTML = q.h2;
        document.getElementById('t3').innerHTML = q.h3;
        
        document.querySelectorAll('.hint-card').forEach(c => c.classList.remove('visible'));
        document.getElementById('letter-hint-container').innerHTML = '';
        document.getElementById('answer-box').classList.add('hidden');
        
        const imgEl = document.getElementById('answer-img');
        imgEl.removeAttribute('src'); // Stop any current loading
        
        revealedLettersCount = 0;
        window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function revealHint(n) {
        if (!isQuizActive) return;
        const hint = document.getElementById(`hint-${n}`);
        hint.classList.add('visible');
        centerElement(`hint-${n}`);
    }

    function revealLetter() {
        if (!isQuizActive) return;
        const answer = currentQuestions[qIndex].ans;
        if (revealedLettersCount < answer.length) {
            const char = answer[revealedLettersCount];
            const circle = document.createElement('div');
            circle.className = 'letter-circle';
            circle.innerText = char === ' ' ? '' : char.toUpperCase();
            if (char === ' ') circle.style.border = 'none';
            document.getElementById('letter-hint-container').appendChild(circle);
            revealedLettersCount++;
            centerElement('letter-hint-container');
        }
    }

    async function showAnswer() {
        if (!isQuizActive) return;
        const q = currentQuestions[qIndex];
        document.getElementById('final-answer').innerText = q.ans;
        
        const box = document.getElementById('answer-box');
        box.classList.remove('hidden');
        document.getElementById('loading-spinner').style.display = 'flex';
        
        document.querySelectorAll('.hint-card').forEach(c => c.classList.add('visible'));
        
        try {
            const imgUrl = await fetchImagen(q.query || q.ans);
            const imgEl = document.getElementById('answer-img');
            imgEl.src = imgUrl;
            imgEl.onload = () => {
                document.getElementById('loading-spinner').style.display = 'none';
            };
        } catch (error) {
            handleImageError();
        }

        setTimeout(() => centerElement('answer-box'), 150);
    }

    async function fetchImagen(prompt) {
        const url = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`;
        const cleanPrompt = `High definition educational photo of ${prompt}. Clear subject, bright lighting, center composition, no text.`;
        
        const response = await fetchWithRetry(url, {
            method: "POST",
            body: JSON.stringify({
                instances: { prompt: cleanPrompt },
                parameters: { sampleCount: 1 }
            })
        });

        const data = await response.json();
        if (data.predictions && data.predictions[0].bytesBase64Encoded) {
            return `data:image/png;base64,${data.predictions[0].bytesBase64Encoded}`;
        }
        throw new Error("Gen failed");
    }

    async function fetchWithRetry(url, options, retries = 5) {
        let delay = 1000;
        for (let i = 0; i < retries; i++) {
            try {
                const response = await fetch(url, options);
                if (response.ok) return response;
            } catch (e) {}
            await new Promise(r => setTimeout(r, delay));
            delay *= 2;
        }
        throw new Error("Final fail");
    }

    function handleImageError() {
        // Safety check to ensure we have a question loaded
        if (!currentQuestions || !currentQuestions[qIndex]) {
            document.getElementById('loading-spinner').style.display = 'none';
            return;
        }
        
        const q = currentQuestions[qIndex];
        const fallback = `https://loremflickr.com/800/600/${encodeURIComponent(q.ans.toLowerCase())}?lock=${qIndex}`;
        
        const imgEl = document.getElementById('answer-img');
        if (imgEl.src !== fallback) {
            imgEl.src = fallback;
        }
        document.getElementById('loading-spinner').style.display = 'none';
    }

    function nextQuestion() {
        if (!isQuizActive) return;
        qIndex = (qIndex + 1) % currentQuestions.length;
        loadQuestion();
    }

    function updateScore(team, val) {
        teamScores[team] += val;
        const el = document.getElementById(`score${team}`);
        el.innerText = teamScores[team];
        el.classList.add('score-pop');
        setTimeout(() => el.classList.remove('score-pop'), 500);
        centerElement('score-area');
    }

    function goHome() {
        isQuizActive = false;
        document.getElementById('grade-picker').classList.remove('hidden');
        document.getElementById('quiz-ui').classList.add('hidden');
        document.getElementById('grade-label').innerText = 'WELCOME';
        document.getElementById('grade-label').className = 'px-8 py-3 rounded-2xl text-white font-black text-xl bg-slate-400 shadow-inner';
        window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    window.addEventListener('keydown', (e) => {
        const key = e.key.toLowerCase();
        if (isQuizActive) {
            if (key === '1') revealHint(1);
            if (key === '2') revealHint(2);
            if (key === '3') revealHint(3);
            if (key === 'l') revealLetter();
            if (key === ' ' || key === 'a') {
                e.preventDefault(); 
                showAnswer();
            }
            if (key === 'n') nextQuestion();
        } else {
            if (key === '1') setGrade('JHS1');
            if (key === '2') setGrade('JHS2');
            if (key === '3') setGrade('JHS3');
        }
        if (key === 'h') goHome();
    });

    lucide.createIcons();
