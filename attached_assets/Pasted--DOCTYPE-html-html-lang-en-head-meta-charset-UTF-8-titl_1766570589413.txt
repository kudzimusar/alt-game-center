    let currentSet = [];

    function startQuiz(grade) {
        // Shuffle the questions for variety
        currentSet = [...data[grade]].sort(() => Math.random() - 0.5);
        qIndex = -1; 
        const badge = document.getElementById('grade-display');
        badge.innerText = "Level: " + grade;
        badge.style.background = `var(--${grade.toLowerCase()})`;
        nextQ();
    }

    function reveal(n) {
        if (qIndex === -1) return;
        document.getElementById('h'+n).classList.add('visible');
    }

    function showAnswer() {
        if (qIndex === -1) return;
        const q = currentSet[qIndex];
        document.getElementById('answer-area').style.display = 'block';
        document.getElementById('ans-text').innerText = q.ans;
        // Use a more reliable image source
        document.getElementById('ans-img').src = `https://loremflickr.com/800/600/${encodeURIComponent(q.ans)}`;
    }

    function nextQ() {
        if (currentSet.length === 0) return alert("Please select a grade first!");
        
        qIndex++;
        if (qIndex >= currentSet.length) {
            alert("End of set! Shuffling again...");
            qIndex = 0;
            currentSet.sort(() => Math.random() - 0.5);
        }

        // Reset UI
        document.getElementById('answer-area').style.display = 'none';
        for(let i=1; i<=3; i++) {
            document.getElementById('h'+i).classList.remove('visible');
            document.getElementById('h'+i).querySelector('.content').innerText = currentSet[qIndex]['h'+i];
        }
    }

    // Keyboard Shortcuts
    window.addEventListener('keydown', (e) => {
        const key = e.key.toLowerCase();
        if(key === '1') reveal(1);
        if(key === '2') reveal(2);
        if(key === '3') reveal(3);
        if(key === 'a' || e.key === 'Enter') showAnswer();
        if(key === 'n' || e.key === ' ') {
            e.preventDefault(); // Stop spacebar scrolling
            nextQ();
        }
    });
