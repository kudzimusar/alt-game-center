        let currentTopic = 'flight';
        let scores = {A: 0, B: 0};
        let lastIndex = -1;
        const GRID_SIZE = 15;

        function initGame() {
            renderTopicMenu();
            generatePuzzle();
        }

        function renderTopicMenu() {
            const menu = document.getElementById('topicMenu');
            menu.innerHTML = '';
            Object.keys(topics).forEach(key => {
                const btn = document.createElement('button');
                btn.className = `btn-topic px-3 py-1 bg-slate-800 rounded-lg font-bold border border-slate-700 flex items-center gap-2 text-xs md:text-sm ${currentTopic === key ? 'active' : ''}`;
                btn.innerHTML = `<span>${topics[key].icon}</span> ${topics[key].label}`;
                btn.onclick = () => switchTopic(key);
                menu.appendChild(btn);
            });
        }

        function switchTopic(key) {
            currentTopic = key;
            renderTopicMenu();
            generatePuzzle();
            document.getElementById('topicTag').innerText = topics[key].label;
            document.getElementById('displayArea').innerText = `New ${topics[key].label} mission loaded. Choose a Grade.`;
            updateTracker();
        }

        function generatePuzzle() {
            const gridBody = document.getElementById('gridBody');
            gridBody.innerHTML = '';
            let grid = Array(GRID_SIZE).fill().map(() => Array(GRID_SIZE).fill(''));
            const words = [...topics[currentTopic].words].sort((a, b) => b.length - a.length);

            words.forEach(word => {
                let placed = false;
                let attempts = 0;
                while (!placed && attempts < 150) {
                    const direction = Math.floor(Math.random() * 2);
                    const row = Math.floor(Math.random() * GRID_SIZE);
                    const col = Math.floor(Math.random() * GRID_SIZE);
                    if (canPlace(grid, word, row, col, direction)) {
                        for (let i = 0; i < word.length; i++) {
                            if (direction === 0) grid[row][col + i] = word[i];
                            else grid[row + i][col] = word[i];
                        }
                        placed = true;
                    }
                    attempts++;
                }
            });

            const letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
            for (let r = 0; r < GRID_SIZE; r++) {
                for (let c = 0; c < GRID_SIZE; c++) {
                    const char = grid[r][c] || letters[Math.floor(Math.random() * letters.length)];
                    const cell = document.createElement('div');
                    cell.className = 'cell';
                    cell.innerText = char;
                    cell.onclick = () => { cell.classList.toggle('selected'); updateTracker(); };
                    gridBody.appendChild(cell);
                }
            }
        }

        function canPlace(grid, word, row, col, dir) {
            if (dir === 0) {
                if (col + word.length > GRID_SIZE) return false;
                for (let i = 0; i < word.length; i++) if (grid[row][col + i] !== '' && grid[row][col + i] !== word[i]) return false;
            } else {
                if (row + word.length > GRID_SIZE) return false;
                for (let i = 0; i < word.length; i++) if (grid[row + i][col] !== '' && grid[row + i][col] !== word[i]) return false;
            }
            return true;
        }

        function updateTracker() {
            const selected = Array.from(document.querySelectorAll('.cell.selected')).map(c => c.innerText).join('');
            document.getElementById('wordTracker').innerText = selected;
        }

        function triggerClue(grade) {
            const list = topics[currentTopic].clues[grade];
            let idx;
            do { idx = Math.floor(Math.random() * list.length); } while (idx === lastIndex && list.length > 1);
            lastIndex = idx;
            const text = list[idx];
            document.getElementById('currentTag').innerText = `${topics[currentTopic].label} - Grade ${grade}`;
            document.getElementById('displayArea').innerHTML = `"${text.replace(/\*\*(.*?)\*\*/g, '<span class="text-pilot-gold font-black underline">$1</span>')}"`;
            const box = document.querySelector('.announcement-box');
            box.style.borderLeftColor = '#fbbf24';
            setTimeout(() => box.style.borderLeftColor = '#0ea5e9', 400);
        }

        function clearHighlights() {
            document.querySelectorAll('.cell').forEach(c => c.classList.remove('selected'));
            updateTracker();
        }

        function resetScores() {
            scores = {A: 0, B: 0};
            document.getElementById('scoreA').innerText = 0;
            document.getElementById('scoreB').innerText = 0;
        }

        window.addEventListener('keydown', (e) => {
            const key = e.key.toLowerCase();
            if (key === 'a') { scores.A++; document.getElementById('scoreA').innerText = scores.A; }
            if (key === 'b') { scores.B++; document.getElementById('scoreB').innerText = scores.B; }
            if (['1','2','3'].includes(key)) triggerClue(parseInt(key));
        });
